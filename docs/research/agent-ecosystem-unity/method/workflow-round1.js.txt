function chunk(arr, n) {
  const out = []
  for (let i = 0; i < arr.length; i += n) out.push(arr.slice(i, i + n))
  return out
}

function oneLine(s) {
  return String(s).replace(/\s+/g, ' ').trim()
}

function cell(s) {
  return oneLine(s).replace(/\|/g, '/')
}

Return at most ' + CLAIMS_PER_ANGLE + ' claims, ranked by how much each would change a decision about whether one shared runtime directory can rely on a SUPPORTED mechanism instead of symlink workarounds. For each claim give: text (one self-contained sentence naming the vendor or standard), vendor (short label), url (the exact page you read), title, publisher, published_date (YYYY-MM-DD or "unknown"), locator (section heading or "page"), source_type, primary_source, and quote (verbatim, at most 300 characters, copied from the fetched text; for an absence claim write the exact page and what you searched it for).',

function skepticPrompt(c, lens) {
  return [
    'ROLE: INDEPENDENT VERIFIER. Lens: ' + lens.title + '.',
    RULES,
    'CLAIM: ' + c.text,
    'CITED URL: ' + c.url,
    'QUOTE THE EXTRACTOR GAVE: ' + c.quote,
    'YOUR LENS: ' + lens.instruction,
    'PROCEDURE: you MUST WebFetch the cited URL before voting. Set fetched=true only if you actually retrieved the page; if the fetch fails, set fetched=false and verdict "unverified". Verdicts: "supported" means you located the quote or an equivalent passage at that source and it says what the claim says (for an absence claim: you re-read the page and it truly does not mention the thing). "contradicted" means the source says something different, or a newer or primary source you found says otherwise; name it in counter_url. "unverified" means the fetch failed, the page changed beyond recognition, or the claim is not locatable. Do NOT vote "contradicted" merely because you are unsure, or because the claim concerns events after your training: lack of confirmation is "unverified". Budget: at most 3 WebFetch and 2 WebSearch calls. Return a reason of at most 300 characters and the passage you located (verbatim, at most 300 characters) in quote if any.',
  ].join('\n\n')
}

function tally(votes) {
  const valid = votes.filter(Boolean)
  const supported = valid.filter(function (v) { return v.verdict === 'supported' && v.fetched === true }).length
  const contradicted = valid.filter(function (v) { return v.verdict === 'contradicted' && v.fetched === true }).length
  let outcome = 'unverified'
  if (contradicted >= 2) outcome = 'killed'
  else if (supported >= 2 && contradicted === 0) outcome = 'supported'
  else if (supported >= 2 && contradicted === 1) outcome = 'disputed'
  return { supported: supported, contradicted: contradicted, unverified: valid.length - supported - contradicted, voters: valid.length, outcome: outcome }
}

async function searchStage(item, a) {
  const r = await agent(searchPrompt(a), {
    phase: 'Search', label: 'search:' + a.id, model: 'sonnet', effort: 'low', agentType: 'Explore', schema: SEARCH_SCHEMA,
  })
  if (!r || !r.results || !r.results.length) {
    dropped.push({ kind: 'search-empty', angle: a.id })
    log('search produced nothing for ' + a.id)
    return null
  }
  if (r.thin) log('thin primary-source coverage for ' + a.id)
  return r
}

async function fetchStage(s, a) {
  if (!s) return null
  const f = await agent(fetchPrompt(a, s), {
    phase: 'Fetch', label: 'fetch:' + a.id, model: 'sonnet', effort: 'medium', agentType: 'Explore', schema: FETCH_SCHEMA,
  })
  if (!f || !f.claims || !f.claims.length) {
    dropped.push({ kind: 'fetch-empty', angle: a.id })
    log('fetch produced no claims for ' + a.id)
    return null
  }
  if (f.claims.length > CLAIMS_PER_ANGLE) {
    dropped.push({ kind: 'claims-over-cap', angle: a.id, count: f.claims.length - CLAIMS_PER_ANGLE })
  }
  return {
    claims: f.claims.slice(0, CLAIMS_PER_ANGLE),
    fetched: f.fetched || [],
    negatives: s.negative_searches || [],
    queries: s.queries_run || [],
    thin: !!s.thin,
  }
}

async function verifyStage(f, a) {
  if (!f) return null
  const verified = await parallel(
    f.claims.map(function (c, ci) {
      return async function () {
        const votes = await parallel(
          LENSES.map(function (lens) {
            return function () {
              return agent(skepticPrompt(c, lens), {
                phase: 'Verify', label: 'verify:' + a.id + ':' + (ci + 1) + ':' + lens.id, model: 'sonnet', effort: 'medium', agentType: 'Explore', schema: VERDICT_SCHEMA,
              })
            }
          }),
        )
        const tagged = votes.map(function (v, i) {
          return v ? Object.assign({ lens: LENSES[i].id }, v) : null
        })
        return Object.assign({}, c, { angle: a.id, votes: tagged, tally: tally(votes) })
      }
    }),
  )
  return {
    angle: a.id,
    claims: verified.filter(Boolean),
    fetched: f.fetched,
    negatives: f.negatives,
    queries: f.queries,
    thin: f.thin,
  }
}

}

const sources = []
const srcByUrl = new Map()
function tierFor(c) {
  if (c.primary_source && (c.source_type === 'official_documentation' || c.source_type === 'specification')) return 'A'
  if (c.primary_source) return 'B'
  if (c.source_type === 'blog' || c.source_type === 'news') return 'C'
  return 'D'
}
function sourceFor(c) {
  if (!srcByUrl.has(c.url)) {
    const s = {
      id: 'S' + (sources.length + 1), url: c.url, title: oneLine(c.title || c.url), publisher: oneLine(c.publisher || 'unknown'),
      published: oneLine(c.published_date || 'unknown'), locator: oneLine(c.locator || 'page'), source_type: c.source_type,
      primary: !!c.primary_source, tier: tierFor(c), resolvable: true,
    }
    srcByUrl.set(c.url, s)
    sources.push(s)
  }
  return srcByUrl.get(c.url)
}

    L.push('- Withdrawn after fetch-based refutation (' + c.id + '): ' + oneLine(c.text) + ' Counter-evidence: ' + counter)
  })
  contradictions.forEach(function (x) {

  })

  L.push('None')
}

const unknowns = claims.filter(function (c) { return c.tally.outcome === 'unverified' || c.tally.outcome === 'disputed' })
if (unknowns.length || S.gaps.length) {
  unknowns.forEach(function (c) {
    const why = c.votes.filter(Boolean).map(function (v) { return v.reason })[0] || 'no verifier reason recorded'
    L.push('- ' + (c.tally.outcome === 'disputed' ? 'Disputed' : 'Unverified') + ' (' + c.id + '): ' + oneLine(c.text) + ' Reason: ' + oneLine(why))

}
L.push('', '## Source Manifest', '')
sources.forEach(function (s) {

  }).join(', ')
}
